二进制部署Jenkins
```bash
# 下载安装Java
sudo apt update
sudo apt install -y openjdk-21-jre
java -version
# 创建独立的程序目录
sudo mkdir -p /opt/jenkins
sudo chown "$USER":"$(id -gn)" /opt/jenkins
# 下载 WAR 文件

# 确认数据目录权限
id
stat -c '目录所有者 UID:GID = %u:%g' /data/jenkins_home
ls -l /var/run/docker.sock
sudo chown -R "$USER":"$(id -gn)" /data/jenkins_home
# 手动启动Jenkins
cd /opt/jenkins
java \
  -Dhttp.proxyHost=127.0.0.1 \
  -Dhttp.proxyPort=7892 \
  -Dhttps.proxyHost=127.0.0.1 \
  -Dhttps.proxyPort=7892 \
  '-Dhttp.nonProxyHosts=localhost|127.*|[::1]' \
  -jar /opt/jenkins/jenkins.war \
  --httpPort=8080
```

配置 Jenkins 后台服务
`sudo vi /etc/systemd/system/jenkins.service`

```ini
[Unit]
Description=Jenkins Automation Server
After=network-online.target docker.service
Wants=network-online.target

[Service]
Type=simple
User=lawwaikit
WorkingDirectory=/opt/jenkins

Environment="JENKINS_HOME=/data/jenkins_home"
Environment="HTTP_PROXY=http://127.0.0.1:7892"
Environment="HTTPS_PROXY=http://127.0.0.1:7892"
Environment="http_proxy=http://127.0.0.1:7892"
Environment="https_proxy=http://127.0.0.1:7892"
Environment="NO_PROXY=localhost,127.0.0.1,::1"

ExecStart=/usr/bin/java -Dhttp.proxyHost=127.0.0.1 -Dhttp.proxyPort=7892 -Dhttps.proxyHost=127.0.0.1 -Dhttps.proxyPort=7892 "-Dhttp.nonProxyHosts=localhost|127.*|[::1]" -jar /opt/jenkins/jenkins.war --httpPort=8080

Restart=on-failure
RestartSec=5
TimeoutStopSec=60
SuccessExitStatus=143

[Install]
WantedBy=multi-user.target
```
 
 启动并设置自动启动
```
sudo systemctl daemon-reload
sudo systemctl enable --now jenkins

检查状态：
sudo systemctl status jenkins
```


流水线调试
```shell
echo "Hello Jenkins"
echo "Current time:"
date
echo "Current directory:"
pwd
echo "Build number: $BUILD_NUMBER"
echo "Job name: $JOB_NAME"
```

`BUILD_NUMBER` 和 `JOB_NAME` 是 Jenkins 提供的环境变量，可以帮助你理解 Jenkins 如何识别每次构建和对应的任务。
