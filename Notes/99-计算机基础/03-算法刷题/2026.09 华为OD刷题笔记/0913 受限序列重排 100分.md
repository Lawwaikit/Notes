# 递归回溯
[[递归回溯]]

# 我理解后的代码
```python
from collections import Counter

def fact(n):
    # 计算n!
    res = 1
    for i in range(2, n + 1):
        res *= i
    return res

def solve(n, k, arr):

    # 1.先不考虑“相邻位置不能相同”
    # 所有排列方式共有 n! 种（n的阶乘）
    total = fact(n)

    counter = Counter(arr)
    # 2.但是数组里面可能有重复元素
    # 假设数组中不同数字出现的数量为x1,x2,..xm
    # 去重的排列数量为total = n!/(x1!x2!...xm!)
    # 除掉重复元素的阶乘
    for count in counter.values():
        total //= fact(count)
    
    # 3.接下来计算“不合法”的排列
    # 第 k-1 和第 k 个位置放的是同一个数字，k的位置其实对结果无影响
    # 对于数x出现xc次，它能组成不合法排列数为
    #   · 让k和k-1位固定为x， x还剩余xc-2个
    #   · 剩余 n-2 个位置可以继续排列
    #   · 同时需要考虑去重(不管需要考虑x，还需要考虑除x之外其它数)
    #     根据2的规律递推。bad = (n - 2)! /((xc - 2)! y1!...ym!)
    bad = 0
    for x, xc in counter.items():
        if xc < 2:   # 出现次数小于2的数不可能构成非法序列
            continue
        cur = fact(n - 2)    # 固定k-1,k两个位置为 x 后，剩余 n-2 个位置的总排列数
        cur //= fact(xc-2)  # 先除掉(xc-2)! 的排列数
        # 其余的数字y 排列数
        for y, yc in counter.items():
            if y != x:
                cur //= fact(yc)
        
        bad += cur

    return total - bad

n = int(input())
k = int(input())
arr = list(map(int, input().split(' ')))
ans = solve(n, k, arr)
print(ans)

```
