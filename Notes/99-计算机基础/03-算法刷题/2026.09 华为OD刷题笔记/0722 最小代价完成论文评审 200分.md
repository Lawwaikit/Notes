# 递归回溯法
需要从n个数里面列出k个数的组合和排列，都可以使用回溯法
[[递归回溯]]

### 我的代码
```python
def solve(n, m, pap_tea_table, costs):
    # n篇论文
    # m个教师 
    # 论文->老师 映射数组

    # 参与评审教师数量最少时的总费用
    # 若存在多种方案满足教师数最少，则选择总费用最小的方案

    # 老师->论文 映射数组
    tea_pap_table = [[] for _ in range(m)]
    for pap_idx in range(n):
        tea_list = pap_tea_table[pap_idx]
        for tea_idx in tea_list:
            tea_pap_table[tea_idx].append(pap_idx)
    # print(f'{tea_pap_table=}\n\n')


    def backtrack(k, start, tea_num, total_cost):
        # 计划投入老师数量为 k 个时, 递归回溯老师的排列组合,计算
        # start 下一位教师从哪里开始选
        # tea_num 当前已投入的老师数量
        # total_cost 当前已花费的费用
        # print(f'{k=}, {start=}, {tea_num=}, {total_cost=}')

        # 当前投入老师达到上限k
        if tea_num == k:
            # print(f"{tea_list=}, {total_cost=}, {pap_vis=}")
            # 是否评审了所有论文
            if all(x>=1 for x in pap_vis):
                nonlocal ans 
                ans = min(ans, total_cost)
            return

        # 待选择老师 tea_idx
        for tea_idx in range(start, m):
            # 选择 tea_idx 老师 评审

            # 对应论文标记为已评审
            for pap_idx in tea_pap_table[tea_idx]:
                pap_vis[pap_idx] += 1

            # 下一位老师从 tea_idx+1 开始选, 已选择老师数量+1, total_cost+
            backtrack(k, tea_idx+1, tea_num+1, total_cost+costs[tea_idx])
 
            # 回溯:对应论文标记为未评审
            for pap_idx in tea_pap_table[tea_idx]:
                pap_vis[pap_idx] -= 1


    INT_INF = 2**32-1
    ans = INT_INF
    # k=1个老师:  0, 1, 2, 3  组合数:C41=4个
    # k=2个老师:  01, 02, 03, 12, 13, 23  组合数:C42=4*3/2! = 6个
    # k=3个老师:  012, 013, 023, 123  组合数C43=4个
    # 老师数量从 [1..m] 计算费用
    for k in range(1, m+1):
        pap_vis = [0] * n
        backtrack(k, 0, 0, 0)
        if ans != INT_INF:
            return ans
    return 0


n = int(input())
m = int(input())
files = []
for i in range(n):
    files.append(list(map(int, input().split(','))))
costs = list(map(int, input().split(',')))

ans = solve(n, m, files, costs)
print(ans)
```
