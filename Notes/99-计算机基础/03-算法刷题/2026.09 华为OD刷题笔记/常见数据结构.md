## 1. `list` 列表

```python
lst = [1, 2, 3]                     # 创建列表
lst.append(x)                      # 末尾添加元素 x
lst.extend(iterable)               # 末尾添加多个元素
lst.insert(i, x)                   # 在下标 i 处插入 x
lst.remove(x)                      # 删除第一个值为 x 的元素
lst.pop(i)                         # 删除并返回下标 i 的元素；省略 i 则删除最后一个
lst.clear()                        # 清空列表
lst.index(x, start, end)           # 查找 x 第一次出现的下标，可指定范围
lst.count(x)                       # 统计 x 出现次数
lst.sort(key=None, reverse=False)  # 原地排序；key 指定排序依据，reverse 是否降序
lst.reverse()                      # 原地反转
lst.copy()                         # 浅拷贝
len(lst)                           # 元素数量
max(lst)                           # 最大值
min(lst)                           # 最小值
sum(lst)                           # 元素求和
sorted(lst, key=None, reverse=False) # 返回排序后的新列表，不修改原列表
```

## 3. `dict` 字典

```python
d = {"a": 1, "b": 2}           # 创建字典
d[key]                         # 根据 key 获取 value
d[key] = value                 # 新增或修改 key 对应的 value
d.get(key, default=None)       # 获取value;key 不存在时返回 default
d.setdefault(key, default)     # 设置value;key 不存在时设置为default,存在时不设置
d.pop(key, default)            # 删除并返回 key 对应的 value,key 不存在时返回 default
d.popitem()                    # 删除并返回最后插入的键值对
d.keys()                       # 获取所有 key
d.values()                     # 获取所有 value
d.items()                      # 获取所有 (key, value)
d.update(other)                # 使用另一个字典更新当前字典
d.clear()                      # 清空字典
d.copy()                       # 浅拷贝
key in d                       # 判断 key 是否存在
len(d)                         # 键值对数量
```

## 4. `set` 集合

```python
s = {1, 2, 3}                  # 创建集合；空集合使用 set()
s.add(x)                       # 添加元素 x
s.remove(x)                    # 删除 x；不存在会报 KeyError
s.discard(x)                   # 删除 x；不存在也不会报错
s.pop()                        # 随机删除并返回一个元素
s.clear()                      # 清空集合
s.copy()                       # 浅拷贝

s1.union(s2)                   # 并集
s1 | s2                        # 并集

s1.intersection(s2)            # 交集
s1 & s2                        # 交集

s1.difference(s2)              # 差集：s1 中有、s2 中没有
s1 - s2                        # 差集

s1.symmetric_difference(s2)    # 对称差集：只存在于其中一个集合
s1 ^ s2                        # 对称差集

s1.issubset(s2)                # 判断 s1 是否为 s2 的子集
s1 <= s2                       # 判断子集

s1.issuperset(s2)              # 判断 s1 是否为 s2 的超集

x in s                         # 判断元素是否存在
len(s)                         # 元素数量
```

## 5. `str` 字符串

```python
s = "hello"                    # 创建字符串
s[i]                           # 根据下标获取字符
s[start:end:step]              # 字符串切片

s.lower()                      # 转小写
s.upper()                      # 转大写
s.strip(chars=None)            # 删除两端空白或指定字符
s.split(sep=None, maxsplit=-1) # 分割字符串，返回 list
s.rsplit(sep=None, maxsplit=-1)# 从右侧开始分割
sep.join(iterable)             # 使用 sep 连接多个字符串

s.replace(old, new, count=-1)  # 替换字符串
s.find(sub, start, end)        # 查找子串下标；找不到返回 -1
s.index(sub, start, end)       # 查找子串下标；找不到抛异常
s.count(sub, start, end)       # 统计子串出现次数
s.startswith(prefix)           # 判断是否以 prefix 开头
s.endswith(suffix)             # 判断是否以 suffix 结尾

s.isdigit()                    # 判断是否全部为数字
s.isalpha()                    # 判断是否全部为字母
s.isalnum()                    # 判断是否全部为字母或数字

len(s)                         # 字符串长度
```

## 8. 常用内置函数

```python
# 计算
len(obj)                       # 获取长度
max(iterable, key=None)        # 最大值；key 指定比较依据
min(iterable, key=None)        # 最小值；key 指定比较依据
sum(iterable, start=0)         # 求和，start 为初始值
abs(x)                         # 绝对值
pow(x, y, mod=None)            # x 的 y 次方；提供 mod 时计算快速幂取模
round(x, ndigits=None)         # 四舍五入到指定小数位
divmod(a, b)                   # 返回 (商, 余数)

# 排序
sorted(iterable, key=None, reverse=False) # 排序并返回新列表
reversed(seq)                  # 返回反向迭代器

# 迭代
range(start, stop, step)       # 生成整数序列
enumerate(iterable, start=0)   # 遍历时同时获取下标和值
zip(*iterables)                # 将多个可迭代对象按位置组合
map(function, iterable)        # 对每个元素执行 function
filter(function, iterable)     # 保留 function 返回 True 的元素

all(iterable)                  # 是否所有元素都为 True
any(iterable)                  # 是否至少一个元素为 True

# 类型转换
int(x, base=10)                # 转整数；base 指定原数字进制
str(x)                         # 转字符串
list(iterable)                 # 转列表
tuple(iterable)                # 转元组
set(iterable)                  # 转集合
dict(iterable)                 # 转字典
```

## 10. 常用 `operator` 函数

```python
operator.add(a, b)             # a + b
operator.sub(a, b)             # a - b
operator.mul(a, b)             # a * b
operator.truediv(a, b)         # a / b
```
