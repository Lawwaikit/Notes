```python
class Node:
    def __init__(self, value, decay):
        self.value = value
        self.decay = decay


def solve(count, total, values, decays):
    # 每个菌子的加工时间固定为5小时
    # 加工所有菌子的总耗时不能超过给定的总加工时间
    # 若菌子的实际价值衰减至零或负值，则不能再售卖，即不需要参与加工
    # 菌子的实际价值=初始价值-衰减速度×从进入工厂到开始加工的等待时间

    nodes = []
    for i in range(count):
        nodes.append(Node(values[i], decays[i]))
    # 衰减速度大的优先加工
    nodes.sort(key=lambda x: x.decay, reverse=True)

    # 最大加工次数
    maxCnt = min(count, total // 5)

    # dp[i] 加工i个菌子时，最大收益
    dp = [-1]*(maxCnt+1)
    dp[0] = 0
    ans = 0

    # 按序遍历每个菌子
    for node in nodes:
        for j in range(maxCnt, 0, -1):   # 当前菌子作为第j个加工 [1..maxCnt]
            if dp[j-1] == -1:  # 无法从上一个j转移
                continue
            money = node.value - node.decay*(j-1)*5  # 加工此菌子获得的价值
            if money<0:
                continue
            dp[j] = max(
                dp[j],    # 不加工当前菌子
                dp[j-1]+money  # 加工当前菌子
            )
    return max(dp)


count, total = map(int, input().split())
values = list(map(int, input().split()))  # 表示第i个菌子的市场价值,单位:元
decays = list(map(int, input().split()))  # 表示第i个菌子的衰减速度,单位:元/小时

print(solve(count, total, values, decays))

```
