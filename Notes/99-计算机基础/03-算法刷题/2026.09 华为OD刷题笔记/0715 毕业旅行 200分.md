# 知识点
## Dijkstra（迪杰斯特拉）
适用于**单源最短路径问题**


## 理解后代码
```python
import heapq

def solve(n, m, w, lines):
    # n城市数量 m路线数量 w最大预算

    # 出发城市编号为 1，目的地城市编号为 n

    # n*?二维记录cost
    graph = [[] for _ in range(n+1)]
    for line in lines:
        u, v, cost = tuple(line)
        graph[u].append((v, cost))

    INF = 100000
    dist = [INF]*(n+1)
    dist[1] = 0
    hq = [(0, 1)]   # 0费用, 从1城市到1

    while hq:
        cost, u = heapq.heappop(hq)

        if cost > dist[u]:
            # 过时节点
            continue
        
        for v, c in graph[u]:
            if dist[u] + c < dist[v]:
                dist[v] = dist[u] + c
                heapq.heappush(hq, (dist[v], v))

    # 判断预算
    if dist[n] <= w:
        return dist[n]
    return -1


n, m, w = map(int, input().split(' '))

lines=[]
for i in range(m):
    lines.append(list(map(int, input().split(' '))))

ans = solve(n, m, w, lines)
print(ans)
```
