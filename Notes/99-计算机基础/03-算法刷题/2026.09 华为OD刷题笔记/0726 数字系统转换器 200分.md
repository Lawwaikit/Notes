# 长除法
[[进制操作，位运算#六、长除法：超长整数字符串转进制]]


### 我的代码
```python
def char_to_int(char):
    # 36进制下,字符转成对应的数字
    if char.isdigit():
        return ord(char) - ord('0')
    elif char.isalpha():
        return ord(char) - ord('a') + 10
    else:
        raise

def int_to_char(num):
    return '0123456789abcdefghijklmnopqrstuvwxyz'[num]


def solve(num_str, sourceDigits, targetDigits):
    # len(sourceDigits) <= 36 说明最多36进制,由 0-1 a-z 表示
    # targetDigits 同理

    # 使用长除法转换进制
    source = len(sourceDigits)
    target = len(targetDigits)
    # print(f'{source}进制到{target}进制')
    ans = []
    while num_str != '0':
        quotient_chars = [] # 商
        remainder = 0   # 余数
        for ch in num_str:
            num = char_to_int(ch[0])   # 最新一位
            cur = remainder * source + num  # 上一位*源进制 + 最新一位
            # 商数 cur // targetDigits
            quotient = cur // target
            quotient_chars.append(int_to_char(quotient))
            # 余数 cur % targetDigits
            remainder = cur % target
            #  商拼接去掉前导0
        num_str = ''.join(quotient_chars).lstrip('0') or '0'
        ans.append(int_to_char(remainder))

    ans = "".join(reversed(ans)).lstrip('0') or '0'
    return ans


num = input()
sourceDigits = input()
targetDigits = input()
ans = solve(num, sourceDigits, targetDigits)
print(ans)

```
